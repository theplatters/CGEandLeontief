\begin{frame}
\frametitle{Why IO Models?}

\begin{itemize}
  \item IO models can be a useful tool to model growth and employment impact of discretionary investments
  \begin{itemize}
      \item Can be either public or private – important is that expenditures are additional and do not depress other spending categories – increased relevance in the context of socio-ecological transformation
  \end{itemize}
  
  \item Three approaches to IO-modeling
 \begin{itemize}
      \item Marxian-Keynesian: Leontief Inverse – extrapolation of current patterns.
      \item Neoclassical: CGE approach – prices and quantities as a result of perfect competition – finding a new equilibrium.
      \item Evolutionary-empirical: explore the impact of price change based on past observations (elasticity of substitution $\sigma$) - can be married with Leontief models (dynamic substitution, i.e. updating $a_{ij}$ based on some estimate of $\sigma$)
    \end{itemize}
\end{itemize}
\end{frame}

  \begin{itemize}

  \end{itemize}

 \begin{itemize}

      \item Neoclassical: CGE approaches by \cite{baqaeeMacroeconomicImpactMicroeconomic2019,baqaeeSupplyDemandDisaggregated2022} – axiomatic variations \parencite{kapellerModelPlatonismEconomicsClassical2013} that supposedly shift standard models and base them on more 'heterodox' assumptions.

    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{IO Models: basics}
\begin{itemize}
  \item Economy consists of $N$ sectors, the output $y_s$ of sector $s$ either is used by other sectors as intermediaries $x_{su}$\footnote{\textit{s} is for 'source' and \textit{u} is for 'use'.}, or used by consumers $c_u$.
    \begin{equation}
      y_s = \sum_{j=1}^N x_{s u} + c_{s}. 
      \label{eq:1}
    \end{equation}
  \item For a fixed point in time, we can introduce $a_{su}$, such that $a_{su} y_u = x_{su}.$ $a_{su}$ is the demand for $y_s$ induced by $y_u$. 
  \item Equation \eqref{eq:1} becomes
     \begin{equation}
       \mathbf y  = A \mathbf y + c
     \end{equation}
     in matrix form.
 \end{itemize} 
\end{frame}

\begin{frame}
  \frametitle{IO Models: introducing $\Omega$}
  \begin{itemize}
    \item We want to tackle two problems:
      \begin{itemize}
        \item We need to express $A$ in monetary terms.    
        \item Do we want the demand side or the supply side?
      \end{itemize}
    \item We therefore introduce $\omega_{us} := \frac{p_s a_{su}}{p_u}$, a cornerstone of both, Leontief as well as neoclassic approaches.
    \item With $\omega_{us} y_u = \frac{p_s x_{su}}{p_u y_u}$ we see, that $\omega_{us}$ captures the relative contribution of sector $s$ to sector $u$. We define $\Omega_p := (\omega_{us})_{1 \leq u,s \leq N}$
    \item Thus    
      $$ \hbf{p} \mathbf{y}  = \hbf p \Omega_p \mathbf{y} + \mathbf{v} = \Omega_p^T \hbf{p}  \mathbf{y} + \hbf{p} \mathbf{c}$$
 $v$ is value added and $\Omega^T$ gives inputs required for each sector (instead of demand).
  \end{itemize}
\end{frame}

\begin{frame}
\frametitle{IO Models: introducing $\Omega$}
\begin{itemize}
\item{$\Omega_p$ can be complemented by additional info on income and consumption}
\begin{align}
  \label{eq:iomatrix}
  \Omega = \left( \begin{array}{@{}ccc|c@{}}
    \omega_{l1} &\dots &\omega_{ln} &  \\ \hline
    \omega_{11} & \dots &\omega_{1n} & \omega_{1l}\\ 
    \vdots &\ddots &\vdots & \vdots\\ 
    \omega_{n1} & \dots & \omega_{nn} & \omega_{nl}    
    \end{array} \right) = 
    \left( \begin{array}{c|c}
    \Omega_l & \\ \hline
    \Omega_p & \Omega_0\\ 
  \end{array} \right),
\end{align}
where $\omega_{sl}$ gives the consumption share of factor $l$ from sector $s$, and $\omega_{lu}$ factor income share in sector $u$. 

  \item Among other things this leads straight to the classic Leontief formulation, where $\bar c$, final consumption, can be expressed as 
    $$\mathbf y = (I - \Omega^T)^{-1} \mathbf c.$$
\end{itemize}
\end{frame}

\section{Neoclassical IO models (CGE models)}

\begin{frame} 
  \frametitle{CGE model}
  \begin{itemize}
    \item Making consumption and interdependent + flexible prices.
    \item It is assumed production and supply have an equilibrium where consumer utility and profits are maximised and markets clear.
      For our context that means:
      \begin{itemize}
        \item Assume $\mathcal D(u)$ is a constant returns to scale utility aggregator. 
        \item Assume every sector has a production function $y_u(\mathbf x)$.
        \item Production $\bar y$, prices $\bar p$ and consumption $\bar c$ are given with the following conditions
          \begin{itemize}
            \item  $y$ maximised profit 
            \item $c$ maximises utility und budget constraints.
            \item  $ \bar {p_u} \bar{y_u} = \sum_{s=1}^N p_s x_{su} + w_u l_u$ holds for all sectors.
           % this is where \Omeage pops up again.
          \end{itemize}
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}
\frametitle{CGE \& IO: Cobb Douglas and CES production functions }
\begin{itemize}
  \item In a neoclassical perspective we want to plunge $\Omega$ into some optimization framework and thereby 'nest' it.
  \item A nested Cobb-Douglas production function takes the form 
    $$ y_u^{\text{CD}}(\mathbf x,l) = A \left(\left(\prod_{s = 1}^N x_{su}^{\omega_{su}} \right)^\alpha l^{1 - \alpha} \right).$$
    $\alpha \in (0,1)$ is the output elasticty and $A$ is some technology parameter.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{CGE \& IO: Cobb Douglas and CES production functions }
\begin{itemize}
   \item In a neoclassical perspective we want to plunge $\Omega$ into some optimization framework and thereby 'nest' it.

  \item A nested CES production function takes the form 
    $$y_u^{\text{CES}}(\mathbf x,\mathbf l) = A \left(
    \omega_{lu} l^\frac{\varepsilon - 1}{\varepsilon} + (1 - \omega_{lu}) X_u(\mathbf x)^\frac{\varepsilon - 1}{\varepsilon} \right)^\frac{\varepsilon}{\varepsilon - 1},$$
    with 
    $X_u(\mathbf{x}) := \left( \sum_{s = 1}^N \omega_{su} x_{su}^{\frac{\theta - 1}{\theta}} \right)^{\frac{\theta}{\theta - 1}}$
    where $\varepsilon$ is the elasticity of substitution between goods and labour and $\theta$ is the elasticity of substitution between goods.
    \item Note that $\lim_{\varepsilon,\theta \to 1} y_u^{\text{CES}} = y_u^{\text{CD}}$ and $\lim_{\varepsilon,\theta \to 0} y_u^{\text{CES}} = y_u^{\text{Leontief}}$
    \end{itemize}
    \end{frame}

\section{research question and results}

\begin{frame}
\frametitle{Research question and operationalization}
\begin{itemize}
  \item \textbf{Basic idea:} Unrealistic mainstream models are modified to become more realistic and, thereby, more heterodox (epicycles, 'axiomatic variation'). Two examples:
  \item \cite{baqaeeMacroeconomicImpactMicroeconomic2019} propose a CES production with a substitution elasticity for production $\epsilon = 0.001$ -- close to Leontief. However, there are two other elasticities (factor vs. intermediates, consumption).
  \item \cite{baqaeeSupplyDemandDisaggregated2022} allow for labor market slack relaxing capacity constraints.
  \item \textbf{Question:} What difference do these changes make?
  \item \textbf{Context: Climate change and transformation} -- 50 bio € additional investments in German construction sector (\textasciitilde 1.25\% of GDP).
\end{itemize}
\end{frame}

\begin{frame}{Summary}
    \begin{itemize}
        \item Two "epicycles"
            \begin{itemize}
                \item Low elasticities
                \item Labour slack
            \end{itemize}
        \item Axiomatic variations can produce drastically different outcomes.
        \item Capacity tends to matter more, than elasticities.
        \item The impact of one choice of elasticity depends heavily on the choice of the other elasticities.
    \end{itemize}
\end{frame}
