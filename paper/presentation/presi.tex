\begin{frame}
\frametitle{Why IO Models?}

\begin{itemize}
  \item IO models can be a useful tool to model growth and employment impact of discretionary investments
  \begin{itemize}
      \item Can be either public or private – important is that expenditures are additional and do not depress other spending categories – increased relevance in the context of socio-ecological transformation
  \end{itemize}
  
  \item Three approaches to IO-modeling
 \begin{itemize}
      \item Marxian-Keynesian: Leontief Inverse – extrapolation of current patterns
      \item Neoclassical: CGE approach – prices and quantities as a result of perfect competition – finding a new equilibrium
      \item Evolutionary-empirical: explore the impact of price change based on past observations (elasticity of substitution $\sigma$) - can be married with Leontief models (dynamic substitution, i.e. updating $a_{ij}$ based on some estimate of $\sigma$)
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{IO Models: Basics}
\begin{itemize}
  \item Economy consists of $N$ sectors, the output $y_s$ of sector $s$ either is used by other sectors as intermediarys $x_{s \to u}$\footnote{We omit the arrow from now on.}, or used by consumers $c_s$.
    \begin{equation}
    p_sy_s = \sum_{u = 1}^n p_s x_{su} + p_s c_s,
      \label{eq:1}
    \end{equation}
  \item For a fixed point in time, we can introduce $a_{su}$, such that $a_{su} y_u = x_{su}.$ $a_{su}$ is the demand for $y_s$ induced by $y_u$. 
  \item Equation \eqref{eq:1} becomes
     \begin{equation}
       \hbf p \mathbf y  = \hbf p A \mathbf  y + \hbf p \mathbf c
     \end{equation}
     in matrix form.
 \end{itemize} 
\end{frame}

\begin{frame}
  \frametitle{Factors}
  \begin{itemize}
      \item In contrast to intermediary goods, that are produced by some sector, factors can not be produced (in the short run)
      \item We restrict ourself to one factor namely labour. The demand for this factor in sector $i$ will be denoted with $l_i$ and the wages with $w_i$.
      \item Adding more factors  could be relevant for future applications. Possibilities include adding CO² and energy or disaggregatingg labour into more specific types.
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{IO Models: Introducing $\Omega$}
  \begin{itemize}
    \item We want to tackle two problems:
      \begin{itemize}
        \item We need to express $A$ in monetary terms.    
        \item Do we want the demand side or the supply side?
      \end{itemize}
    \item We therefore introduce $\omega_{us} := \frac{p_s a_{su}}{p_u}$.
    \item With $\omega_{ij} y_j = \frac{p_i x_{ij}}{p_j y_j}$ we see, that $\omega_{ij}$ captures the relative contribution of sector $i$ to sector $j$.
    \item Thus    
      $$\hbf p \mathbf y = \Omega^T \hbf p \mathbf y + \hbf p \mathbf c$$
  \end{itemize}
\end{frame}
\section{Leontief model}
\begin{frame} 
  \frametitle{Leontief model}
  \begin{itemize}
    \item  We assume production is so, that costs are minimal i.e intermediaries are given as the solution to
      \begin{gather}
          \label{eq:costmin}
          \min_{\mathbf x_{\mathbf s, u} \in \mathbb R^N_+} \sum_{s=1}^n p_s x_{s,u} \\
          \text{subject to }  y_s(\mathbf x) = \sum_{u=1}^N x_{su} + \bar c_ s \quad \text{for all } u=s
      \end{gather}
    and consumption is exogenuously given.
    \item The Leontief production function takes the form $y_u(\mathbf x) = \min_{s}(\frac{x_{su}}{a_{us}})$. Outputs are based on a fixed proportion of inputs.
    \item Thus Intermediary demand is given as $x_{su} = a_{su} y_u$.
    \item Prices are assumed to be consant. We shall not worry about them for now.
  \end{itemize}
\end{frame}

\begin{frame}
\frametitle{Leontief Model}
\begin{itemize}
  \item For a given consumption vector $\bar c$, the required production is then given as 
    $$\hbf p \mathbf y = (I - \Omega^T)^{-1}  \hbf p \mathbf c$$
  \item This model can be easily extended to include factors, by expanding $\Omega$ and $y$:
\begin{align}
  \label{eq:iomatrix}
  \Omega = \left( \begin{array}{@{}ccc|c@{}}
    \omega_{01} &\dots &\omega_{0n} &  \\ \hline
    \omega_{11} & \dots &\omega_{1n} & \omega_{1l}\\ 
    \vdots &\ddots &\vdots & \vdots\\ 
    \omega_{n1} & \dots & \omega_{nn} & \omega_{nl}    
    \end{array} \right) = 
    \left( \begin{array}{c|c}
    \Omega_0 & \\ \hline
    \Omega_p & \Omega_l\\ 
  \end{array} \right).
\end{align}
$\omega_{0i}$ gives the consumption share of factor $0$ from sector $i$, and $\omega_{il}ω$ factor share in sector $i$. 
\end{itemize}
\end{frame}

\section{CGE Model}

\begin{frame} 
  \frametitle{CGE Model}
  \begin{itemize}
    \item Making consumption and production mutually independent.
    \item It is assumed production and supply have an equilibrium where consumer utility and profits are maximised and markets clear.
      For our context that means:
      \begin{itemize}
        \item Assume $\mathcal D(u)$ is a constant returns to scale utility aggregator. 
        \item Assume every sector has a production function $y_i(\mathbf x)$.
        \item Production $\bar y$, prices $\bar p$ and consumption $\bar c$ are given with the following conditions
          \begin{itemize}
            \item  $y$ maximised profit 
            \item $c$ maximises utility und budget constraints
            \item  $ \bar {p_i} \bar{y_i} = \sum_{j=1}^N p_j x_{ji} + w_i l_i$ holds for all sectors
          \end{itemize}
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Cobb Douglas and CES production functions }
\begin{itemize}
  \item A Cobb-Douglas production function takes the form 
    $$y(\mathbf x,l) = A \left(x^\alpha l^{1 - \alpha} \right).$$ 
    $\alpha \in (0,1)$ is the output elasticty and $A$ is some technology parameter.
  \item A CES production function takes the form 
    $$y_u(x) = \left(\sum_{s=1}^N \omega_{su} x_{su}^{\frac{\epsilon - 1}{\epsilon}} \right)^{\frac{\epsilon}{1 - \epsilon}},$$ 
    where $\epsilon$ is the elasticity of substitution.
\end{itemize}
\end{frame}

\begin{frame}
  \frametitle{CES and Elasticities}
  \begin{itemize}
    \item In \cite{baqaeeMacroeconomicImpactMicroeconomic2019} a CGE framework is used, however the production function takes an interresting form.
    \item $$y_u(l_u, \mathbf x_{\mathbf s,u}) =  A_u \left(\omega_{lu} l_u^{\frac {\theta-1}{\theta}} + (1- \omega_{lu}) X^{\frac{\theta-1}{\theta}}(\mathbf x_{\mathbf s,u}) \right)^{\frac{\theta}{\theta - 1}},$$
    with 
  \[
    X(\mathbf x_{\mathbf s, u}) = \left(\sum_{s=1}^N \omega_{su} x_{su}^{\frac{\varepsilon - 1}{\varepsilon}} \right)^{\frac{\varepsilon}{\varepsilon - 1}},
  \]
  where $\varepsilon$ is the substitution elasticity of intermediary goods and $\theta$ is the substitution elasticty between labour and goods.
  \end{itemize}
\end{frame}


\begin{frame}
\frametitle{What Makes this Special?}
\begin{itemize}
  \item \cite{baqaeeMacroeconomicImpactMicroeconomic2019} propose $\epsilon = 0.001$ and $\theta = 0.5$.
  \item The limit of a CES function, when the substitution elasticty goes to $0$ is a Leontief production function.
    The limit of $\epsilon \to 1$ is a Cobb-Douglas production function.
  \item The small value of $\epsilon$ reflects an introduction of assumptions found in Marxist/Keynsian economics into neoclassic literature.
\end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Labour Market Slack?}
  \begin{itemize}
    \item In CGE models, like in \cite{baqaeeMacroeconomicImpactMicroeconomic2019} factors are assumed to have limited supply, that is fully utilised in an equilibrium.
    \item In contrast, the Leontief model assumes factors are abundant.
  \item We extended the model to allow for labour market slack, so some additional supply of factor $l$ can be allocated.
   \end{itemize}
\end{frame}
